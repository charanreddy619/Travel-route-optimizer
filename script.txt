const N={A:[50,150],B:[180,50],C:[180,250],D:[360,50],E:[470,200]};      // SVG positions
const X={A:[0,0],B:[2,1],C:[2,0],D:[7,1],E:[9,1]};                        // km coordinates for the A* heuristic
const R=[['A','B',4],['A','C',2],['B','C',1],['B','D',5],['C','D',8],['C','E',10],['D','E',2]];
const adj={};Object.keys(N).forEach(k=>adj[k]=[]);R.forEach(([a,b,w])=>{adj[a].push([b,w]);adj[b].push([a,w])});
const h=(a,b)=>Math.hypot(X[a][0]-X[b][0],X[a][1]-X[b][1]);
class PQ{constructor(){this.a=[]}get size(){return this.a.length}
 push(p,v){const a=this.a;a.push([p,v]);let i=a.length-1;while(i&&a[i][0]<a[(i-1)>>1][0]){const j=(i-1)>>1;[a[i],a[j]]=[a[j],a[i]];i=j}}
 pop(){const a=this.a,t=a[0],l=a.pop();if(a.length){a[0]=l;let i=0;for(;;){let c=2*i+1;if(c+1<a.length&&a[c+1][0]<a[c][0])c++;if(c>=a.length||a[c][0]>=a[i][0])break;[a[i],a[c]]=[a[c],a[i]];i=c}}return t}}
function solve(s,d,star){const g={},prev={},done=new Set(),order=[],q=new PQ();Object.keys(N).forEach(k=>g[k]=Infinity);g[s]=0;q.push(star?h(s,d):0,s);
 while(q.size){const[,u]=q.pop();if(done.has(u))continue;done.add(u);order.push(u);if(u===d)break;
  for(const[v,w]of adj[u]){const n=g[u]+w;if(n<g[v]){g[v]=n;prev[v]=u;q.push(n+(star?h(v,d):0),v)}}}
 const path=[];if(g[d]<Infinity)for(let n=d;n;n=prev[n])path.unshift(n);return{path,total:g[d],order}}
